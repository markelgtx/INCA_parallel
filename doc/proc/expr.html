    function expr(x,y,z,xa,ya,za,npr) 
        implicit none
        double precision :: expr
        double precision, intent(in) :: x, y, z,xa,ya,za
        integer, intent(in) :: npr
        expr=dexp(-Alpha(npr)*((x-xa)**2.d0+(y-ya)**2.d0+(z-za)**2.d0))
    end function expr
