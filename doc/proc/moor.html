    function MoOr(x,y,z,mo) 
        implicit none
        double precision :: MoOr
        double precision, intent(in) :: x, y, z
        integer, intent(in) :: mo
        integer :: i
        MoOr = 0.d0
        do i=1,nprim 
            if (dabs(T(mo,i)) > 1.d-10) then
                MoOr = MoOr + T(mo,i) * Prim(x,y,z,i)
            end if 
        end do
    end function MoOr
