   subroutine cmatrix()   
   use wfxinfo
   use loginfo
   use geninfo
   implicit none
   integer :: i, j, k
        
   allocate(Ckalk(noccmo,nao))
   write(*,*) nao
   write(*,*) noccmo
   do i=1,noccmo
    k=1
    do j=1,nao
         Ckalk(i,j)=T(i,k)/(Flg(k)) 
         k=k+npao(j)  
     end do
   end do   
   
   end subroutine cmatrix
