    function MO_a(x,y,z,mo) 
        implicit none
        double precision :: MO_a
        double precision, intent(in) :: x, y, z
        integer, intent(in) :: mo
        integer :: i
        MO_a = 0.d0
        do i=1,nprim 
            if (dabs(T_a(mo,i)) > 1.d-10) then
                MO_a = MO_a + T_a(mo,i) * Prim(x,y,z,i)
            end if 
        end do
    end function MO_a
