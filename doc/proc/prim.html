    function Prim(x,y,z,npr) 
        implicit none
        double precision :: Prim
        double precision, intent(in) :: x, y, z
        integer, intent(in) :: npr
        double precision :: xa, ya, za
        xa = cartes(Ra(npr),1)
        ya = cartes(Ra(npr),2)
        za = cartes(Ra(npr),3)
    
        Prim=(x-xa)**(dble(TMN(npr,1)))* &
             (y-ya)**(dble(TMN(npr,2)))* &
             (z-za)**(dble(TMN(npr,3)))* &
             dexp(-Alpha(npr)*((x-xa)**2.d0+(y-ya)**2.d0+(z-za)**2.d0))               
    end function Prim
